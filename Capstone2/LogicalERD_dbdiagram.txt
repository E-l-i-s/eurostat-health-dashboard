Logical Data Model — 3NF — dbdiagram.io Syntax
================================================
Physical Activity and Chronic Disease Burden Across Europe

//==============================================================
// ENTITY: Country
//==============================================================
Table Country {
  country_code  varchar(2)  [pk, not null, note: "ISO 3166-1 alpha-2"]
  country_name  varchar(50) [not null, note: "Full country name in English"]
  region        varchar(30) [note: "European region grouping (e.g. Western, Eastern, Southern, Northern)"]
}

//==============================================================
// ENTITY: AgeGroup
//==============================================================
Table AgeGroup {
  age_code    varchar(10) [pk, not null, note: "EHIS age group code (TOTAL, Y18-24, ...)"]
  label       varchar(30) [not null, note: "Human-readable label"]
  lower_bound integer     [note: "Lower bound of age range in years (NULL for TOTAL)"]
  upper_bound integer     [note: "Upper bound of age range in years (NULL for TOTAL or open-ended)"]
}

//==============================================================
// ENTITY: Sex
//==============================================================
Table Sex {
  sex_code varchar(1) [pk, not null, note: "T, M, or F"]
  label    varchar(10) [not null, note: "Total, Male, or Female"]
}

//==============================================================
// ENTITY: IndicatorCategory
//==============================================================
Table IndicatorCategory {
  category_code varchar(30) [pk, not null, note: "Unique category identifier"]
  category_name varchar(50) [not null, note: "Physical Activity, Chronic Disease, or BMI"]
}

//==============================================================
// ENTITY: DataSource
//==============================================================
Table DataSource {
  source_code    varchar(15) [pk, not null, note: "Eurostat dataset code (hlth_ehis_pe3, etc.)"]
  full_name      varchar(80) [not null, note: "Full Eurostat dataset name"]
  api_endpoint   varchar(150) [not null, note: "API URL for retrieval"]
  retrieval_date date        [note: "Date data was last retrieved"]
}

//==============================================================
// ENTITY: HealthIndicator
//==============================================================
Table HealthIndicator {
  indicator_code varchar(30)  [pk, not null, note: "Unique indicator code"]
  indicator_label varchar(100) [not null, note: "Human-readable indicator name"]
  category_code   varchar(30)  [not null, ref: > IndicatorCategory.category_code, note: "FK to IndicatorCategory"]
  source_code     varchar(15)  [not null, ref: > DataSource.source_code, note: "FK to DataSource"]
  unit_of_measure varchar(10)  [not null, note: "PC (percentage) or other unit"]
  description     varchar(255) [note: "Short description of the indicator"]
}

//==============================================================
// ENTITY: SurveyWave
//==============================================================
Table SurveyWave {
  wave_id      integer      [pk, not null, note: "Survey year (2008, 2014, 2019)"]
  description  varchar(50)  [not null, note: "EHIS Wave 1/2/3"]
  year         integer      [not null, note: "Calendar year of data collection"]
}

//==============================================================
// ENTITY: Measurement (FACT TABLE)
//==============================================================
Table Measurement {
  measurement_id  integer      [pk, not null, note: "Auto-increment primary key"]
  country_code    varchar(2)   [not null, ref: > Country.country_code, note: "FK to Country"]
  age_code        varchar(10)  [not null, ref: > AgeGroup.age_code, note: "FK to AgeGroup"]
  sex_code        varchar(1)   [not null, ref: > Sex.sex_code, note: "FK to Sex"]
  indicator_code  varchar(30)  [not null, ref: > HealthIndicator.indicator_code, note: "FK to HealthIndicator"]
  wave_id         integer      [not null, ref: > SurveyWave.wave_id, note: "FK to SurveyWave"]
  value           real         [not null, check: "value >= 0 AND value <= 100", note: "Percentage value"]
  data_suppressed integer      [not null, default: 0, note: "0 = published, 1 = suppressed"]
  is_outlier      integer      [not null, default: 0, note: "0 = not outlier, 1 = outlier-flagged"]
}

//==============================================================
// REFERENCE DEFINITIONS
//==============================================================
Ref: Measurement.country_code > Country.country_code
Ref: Measurement.age_code > AgeGroup.age_code
Ref: Measurement.sex_code > Sex.sex_code
Ref: Measurement.indicator_code > HealthIndicator.indicator_code
Ref: Measurement.wave_id > SurveyWave.wave_id
Ref: HealthIndicator.category_code > IndicatorCategory.category_code
Ref: HealthIndicator.source_code > DataSource.source_code
